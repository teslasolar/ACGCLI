The one file that generates everything

Full Example

# acg.yaml — drop this in any repo root
 
name: my-project
version: 1.0.0
description: "What this repo does"
author: thomas
guild: acg
 
# ═══ RUNTIME ═══
runtime:
language: python # python | node | rust | go | static
entry: src/main.py
env_file: .env
port: 8080
 
# ═══ COMMANDS ═══
commands:
run:
cmd: "python src/main.py"
description: "Start the application"
test:
cmd: "pytest tests/ -v"
description: "Run test suite"
build:
cmd: "python -m build"
description: "Build distribution"
deploy:
cmd: "./scripts/deploy.sh"
description: "Deploy to production"
env: production # sets NODE_ENV
confirm: true # requires --yes
lint:
cmd: "ruff check src/"
description: "Lint source code"
clean:
cmd: "rm -rf dist/ build/ __pycache__/"
description: "Clean build artifacts"
 
# ═══ CUSTOM COMMANDS ═══
custom:
crawl:
cmd: "python spiders/crawl.py --all"
description: "Run all spiders"
 
# ═══ HEALTH CHECKS ═══
health:
- name: "Python version"
check: "python --version"
expect: "3.10"
- name: "Tests passing"
check: "pytest tests/ -q"
 
# ═══ API CONFIG ═══
api:
port: 9000
cors: true
 
# ═══ MCP CONFIG ═══
mcp:
name: my-project-mcp
extra_tools:
- name: search_index
description: "Search the index"
params: { query: string }
cmd: python scripts/search.py --query "{query}"
 
# ═══ AI CONFIG ═══
ai:
provider: local # local | anthropic | openai
model: phi3
context_files:
- README.md
- acg.yaml

Required Fields

name: # project name (required)
runtime: # runtime config (required)
language: # python | node | rust | go | static
commands: # at least one command (required)

Command Properties

cmd: # shell command to run (required)
description: # human-readable description
env: # sets NODE_ENV (e.g. "production")
confirm: # true = requires --yes flag or interactive Y
watch: # glob pattern for auto-restart (future)
